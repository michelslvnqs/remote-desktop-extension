const SHEET_ID = '1bH1MC8jU6D7oMsqDdZlWRJBLp6vcJlqey8hP_W89pBQ';
const SHEET_NAME = 'Sheet1';

function doPost(e) {
  try {
    const params = e.postData ? JSON.parse(e.postData.contents) : {};
    const code = (params.code || params.text || '').toString().trim();
    const timestamp = new Date().toISOString();
    
    if (!code) {
      return ContentService
        .createTextOutput(JSON.stringify({ success: false, error: 'Código não fornecido' }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    const ss = SpreadsheetApp.openById(SHEET_ID);
    let sheet = ss.getSheetByName(SHEET_NAME);
    
    if (!sheet) {
      sheet = ss.insertSheet(SHEET_NAME);
      sheet.appendRow(['Codigo', 'Timestamp']);
    }
    
    if (sheet.getLastRow() === 0 || sheet.getRange(1, 1).getValue() !== 'Codigo') {
      sheet.clear();
      sheet.appendRow(['Codigo', 'Timestamp']);
    }
    
    sheet.appendRow([code, timestamp]);
    
    return ContentService
      .createTextOutput(JSON.stringify({
        success: true,
        code: code,
        timestamp: timestamp,
        message: 'Código salvo com sucesso'
      }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ success: false, error: error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    let sheet = ss.getSheetByName(SHEET_NAME);
    
    if (!sheet) {
      return ContentService
        .createTextOutput(JSON.stringify({ success: true, codes: [] }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    const data = sheet.getDataRange().getValues();
    const codes = [];
    
    for (let i = 1; i < data.length; i++) {
      if (data[i][0]) {
        codes.push({
          code: data[i][0],
          timestamp: data[i][1] || 'N/A'
        });
      }
    }
    
    return ContentService
      .createTextOutput(JSON.stringify({ success: true, codes: codes }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ success: false, error: error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
